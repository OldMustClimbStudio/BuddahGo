    private static void PhysicsSample(string kind,uint clientReplayTick,uint serverReplayTick)
    {
        if(writer==null)return;
        foreach(var motor in Motors())
        {
            var rb=motor.GetComponent<Rigidbody>();
            Record(new{kind,owner=motor.OwnerId,localOwner=motor.IsOwner,
                localTick=motor.TimeManager.LocalTick,serverTick=motor.TimeManager.Tick,clientReplayTick,serverReplayTick,
                position=Vec(rb.position),velocity=Vec(rb.velocity),time=Time.realtimeSinceStartupAsDouble});
        }
    }

    {
        // Leave the natural start untouched; sample its first seconds before fixture setup.
        if(now-unlocked<4)return;

            if(Arg("-reviewMode")=="visual")matrixIndex=11;

            testInputs.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
            InputSystem.settings=testInputs;testKeyboard=InputSystem.AddDevice<Keyboard>();

            {
                TrackPositions[motor.OwnerId]=motor.transform.position;TrackRotations[motor.OwnerId]=motor.transform.rotation;

            }

        {
            matrixIndex++;matrixBegan=now;
            if(matrixIndex>=total){File.WriteAllText(controlPath,"{\"index\":"+total+"}");return;}
            int owner=matrixIndex>=12?matrixIndex-12:matrixIndex/6;string mode=matrixIndex>=12?"charged":Cases[matrixIndex%6];
            if(Arg("-reviewMode")=="push"){owner=matrixIndex;mode="push";}
            floor.SetActive(mode!="respawn");
            foreach(var executor in UnityEngine.Object.FindObjectsByType<SkillExecutor>(FindObjectsSortMode.None))
            {
                executor.ResetActiveSkillEffectsServer();
                var obs=executor.GetComponent<ObsessionFigure>();
                Set(obs,"backfirePMaxPercent",executor.OwnerId==owner&&mode=="burst"?100f:0f);
                Set(obs,"backfireKeepPMaxAtOrAboveX",0f);Set(executor,"_castLockedUntil",0f);
                Array.Clear((float[])Read(executor,"_nextReadyTime"),0,SkillLoadout.SlotCount);
                executor.GetComponent<SkillLoadout>().SetSlotsServer(new[]{"push_projectile_hands","giant","acceleration"});

                    hands.ActivateProjectilePushMode(12,0,120,3,100,new Vector3(10,10,12),0,0,null,null,.25f,false,true);
                Vector3 offset=executor.OwnerId==owner?Vector3.zero:matrixIndex>=12?new Vector3(200,0,20):
                    mode=="push"?new Vector3(-4.1f,0,10):mode=="ordinary"?new Vector3(-4.1f,0,40):
                    mode=="collision"?new Vector3(0,0,30):new Vector3(0,0,20);

        JObject item;

    private static void StopMatrix()
    {
        if(testKeyboard!=null)InputSystem.RemoveDevice(testKeyboard);
        if(originalInputs!=null)InputSystem.settings=originalInputs;
        if(testInputs!=null)UnityEngine.Object.DestroyImmediate(testInputs);
    }
    public static void RecordCollision(BuddahPredictedMotor motor,Collision collision)
    {
        var other=collision.collider.GetComponentInParent<BuddahPredictedMotor>();
        if(other==null)return;
        Record(new{kind="actorCollision",index=observedIndex,owner=motor.OwnerId,other=other.OwnerId,
            localTick=motor.TimeManager.LocalTick,serverTick=motor.TimeManager.Tick,
            impulse=Vec(collision.impulse),relativeVelocity=Vec(collision.relativeVelocity)});
    }
